```html
<view class="stage-wrap">
  <view class="stages">
    <view
      v-for="(t, i) in TAB_OPTS"           <!-- 或原 pills 数组 -->

  </view>
  <!-- 二级筛选面板：chips 浮层 -->
  <view v-if="panel === 'all'" class="field-panel" :class="{ closing }">
    <view class="p-group">组名</view>

      <text class="p-chip" :class="{ act: field === '' }" @tap="pickField('')">全部</text>
      <text v-for="f in FIELDS" :key="f" class="p-chip" :class="{ act: field === f }" @tap="pickField(f)">{{ f }}</text>
    </view>
  </view>
</view>
<!-- 蒙层：从 tab 底开始置灰 -->
<view v-if="panel" class="panel-mask" :style="{ top: maskTop + 'px' }" @tap="startClosePanel" />
```

### JS 模式

```js
const panel = ref('')
const closing = ref(false)
const maskTop = ref(0)
let panelCloseT = null
const PANEL_CLOSE_MS = 210

const startClosePanel = () => {
  if (closing.value) return
  closing.value = true
  clearTimeout(panelCloseT)
  panelCloseT = setTimeout(() => { panel.value = ''; closing.value = false; panelCloseT = null }, PANEL_CLOSE_MS)
}
const togglePanel = () => {
  if (panel.value === 'all') { startClosePanel(); return }
  clearTimeout(panelCloseT); panelCloseT = null; closing.value = false
  panel.value = 'all'
  uni.nextTick(() => {
    uni.createSelectorQuery().select('.stage-wrap').boundingClientRect((rect) => {
      if (rect && rect.bottom) maskTop.value = Math.round(rect.bottom)
    }).exec()
  })
}
// 方案 A：非全部 tab 再点取消；全部 tab 未停先清筛、停下再开面板；▾ 独立开关
const pickStageTab = (k) => {
  if (k !== 'all') { startClosePanel(); activeX.value = activeX.value === k ? 'all' : k; fetchList(true); return }
  if (activeX.value !== 'all') { startClosePanel(); activeX.value = 'all'; fetchList(true); return }
  togglePanel()
}
```

## 关键注意
1. **删除旧胶囊样式**（.fbar/.fpill/.filter-scroll 等）及对应模板，避免死代码 + 类名冲突
2. **保留**原筛选逻辑：数据过滤函数（switchType/selectCat 等）改名或桥接到新 pick 函数，核心是调 fetchList/filter 时用新状态变量
3. **多维度页面**（成果库 2 组 chips）：面板内可放多个 .p-group；若原页面有多个胶囊各管一个维度但无面板，把次级维度放面板 chips
4. **蒙层 z-index**：若页面的吸顶容器 z-index > 41，把 .panel-mask 的 z-index 设为该容器之下（如 sticky-head z-index:20 时用 15）；若 stage-wrap 自身 sticky（如成果库），保持 41
5. **搜索栏**：保留原搜索框位置与样式，只动筛选条
6. 已对齐的页面（achievements/list、training/courses）不要动
7. 排除清单（用户指定保持现状）：colleges/list、activities/list、exhibitions/list、reports/list、emergency/resources、emergency/depts、pilots/list
